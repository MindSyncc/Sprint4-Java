-- 1. Categoria
CREATE TABLE categoria (
    id_categoria NUMBER GENERATED BY DEFAULT AS IDENTITY,
    tipoCategoria VARCHAR2(50) NOT NULL,
    CONSTRAINT pk_categoria PRIMARY KEY (id_categoria),
    CONSTRAINT uq_categoria_tipo UNIQUE (tipoCategoria),
    CONSTRAINT ck_categoria_tipo CHECK (tipoCategoria IS NOT NULL)
);

-- 2. Insumo
CREATE TABLE insumo (
    id_insumo NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    lote VARCHAR2(50),
    dataValidade DATE,
    unidadeMedida VARCHAR2(20),
    codigo_de_barras VARCHAR2(50),
    id_categoria NUMBER,
    CONSTRAINT fk_insumo_categoria FOREIGN KEY (id_categoria)
        REFERENCES categoria(id_categoria)
);

-- 3. Estoque
CREATE TABLE estoque (
    id_estoque NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    qtdAtual NUMBER(6) NOT NULL,
    qtdMinima NUMBER(6) NOT NULL,
    qtdMaxima NUMBER(6) NOT NULL,
    status VARCHAR2(20) NOT NULL,
    CONSTRAINT chk_quantidades CHECK (qtdMinima > 0 AND qtdMaxima >= qtdMinima),
    CONSTRAINT chk_status CHECK (status IN ('Disponível', 'Baixo', 'Esgotado'))
);

-- 4. Estoque_Insumo
CREATE TABLE estoque_insumo (
    id_estoque NUMBER NOT NULL,
    id_insumo  NUMBER NOT NULL,
    quantidade NUMBER DEFAULT 0,
    CONSTRAINT pk_estoque_insumo PRIMARY KEY (id_estoque, id_insumo),
    CONSTRAINT fk_ei_estoque FOREIGN KEY (id_estoque) REFERENCES estoque(id_estoque),
    CONSTRAINT fk_ei_insumo FOREIGN KEY (id_insumo) REFERENCES insumo(id_insumo)
);

-- 5. Unidade
CREATE TABLE unidade (
    id_unidade NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_estoque NUMBER NOT NULL,
    nome_unidade VARCHAR2(100) NOT NULL,
    rua VARCHAR2(100) NOT NULL,
    numero VARCHAR2(10) NOT NULL,
    bairro VARCHAR2(50) NOT NULL,
    cidade VARCHAR2(50) NOT NULL,
    estado VARCHAR2(2) NOT NULL,
    cep VARCHAR2(10) NOT NULL,
    CONSTRAINT unq_unidade UNIQUE (nome_unidade, rua, numero, bairro, cidade, estado, cep),
    CONSTRAINT chk_estado_unidade CHECK (LENGTH(estado) = 2),
    CONSTRAINT chk_cep_unidade CHECK (REGEXP_LIKE(cep, '^[0-9]{5}-?[0-9]{3}$')),
    CONSTRAINT fk_unidade_estoque FOREIGN KEY (id_estoque) REFERENCES estoque(id_estoque)
);

-- 6. Fornecedores
CREATE TABLE fornecedores (
    id_fornecedor NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome_fornecedor VARCHAR2(200) NOT NULL,
    telefone VARCHAR2(20) NOT NULL,
    email VARCHAR2(100) NOT NULL,
    cnpj VARCHAR2(14) NOT NULL UNIQUE,
    rua VARCHAR2(100) NOT NULL,
    numero VARCHAR2(10) NOT NULL,
    bairro VARCHAR2(50) NOT NULL,
    cidade VARCHAR2(50) NOT NULL,
    estado VARCHAR2(2) NOT NULL,
    cep VARCHAR2(10) NOT NULL,
    CONSTRAINT chk_estado CHECK (LENGTH(estado) = 2),
    CONSTRAINT chk_cep CHECK (REGEXP_LIKE(cep, '^[0-9]{5}-?[0-9]{3}$')),
    CONSTRAINT chk_telefone CHECK (REGEXP_LIKE(telefone, '^[0-9]{8,20}$'))
);

-- 7. Funcionarios
CREATE TABLE funcionarios (
    id_funcionario NUMBER GENERATED BY DEFAULT AS IDENTITY,
    funcional VARCHAR2(8) NOT NULL,
    nome VARCHAR2(200) NOT NULL,
    cpf VARCHAR2(11) UNIQUE NOT NULL,
    dataNascimento DATE NOT NULL,
    salario NUMBER(10,2) NOT NULL,
    dataInicio DATE NOT NULL,
    dataTermino DATE,
    turno VARCHAR2(10) NOT NULL,
    cargo VARCHAR2(50) NOT NULL,
    senhaHash VARCHAR2(64) NOT NULL,
    permissao VARCHAR2(20) NOT NULL,
    rua VARCHAR2(100) NOT NULL,
    numero VARCHAR2(10) NOT NULL,
    bairro VARCHAR2(50) NOT NULL,
    cidade VARCHAR2(50) NOT NULL,
    estado VARCHAR2(2) NOT NULL,
    cep VARCHAR2(10) NOT NULL,
    id_unidade NUMBER NOT NULL,
    CONSTRAINT pk_funcionario PRIMARY KEY (id_funcionario),
    CONSTRAINT uq_funcional UNIQUE (funcional),
    CONSTRAINT chk_funcional CHECK (REGEXP_LIKE(funcional, '^[0-9]{8}$')),
    CONSTRAINT chk_cpf_funcionario CHECK (REGEXP_LIKE(cpf, '^[0-9]{11}$')),
    CONSTRAINT chk_salario CHECK (salario > 0),
    CONSTRAINT chk_datas CHECK (dataTermino IS NULL OR dataTermino >= dataInicio),
    CONSTRAINT chk_estado_funcionario CHECK (LENGTH(estado) = 2),
    CONSTRAINT chk_cep_funcionario CHECK (REGEXP_LIKE(cep, '^[0-9]{5}-?[0-9]{3}$')),
    CONSTRAINT chk_turno CHECK (turno IN ('Manhã', 'Tarde', 'Noite')),
    CONSTRAINT chk_permissao CHECK (permissao IN ('Analista Local', 'Analista Corporativo', 'Almoxarife', 'Colaborador Interno')),
    CONSTRAINT fk_funcionario_unidade FOREIGN KEY (id_unidade) REFERENCES unidade(id_unidade)
);

-- 8. Pedido
CREATE TABLE pedido (
    id_pedido NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome_item VARCHAR2(200),
    quantidade NUMBER(6),
    status VARCHAR2(20),
    data_pedido DATE DEFAULT SYSDATE,
    id_funcionario NUMBER NOT NULL,
    id_fornecedor NUMBER NOT NULL,
    CONSTRAINT fk_pedido_funcionario FOREIGN KEY (id_funcionario)
        REFERENCES funcionarios(id_funcionario),
    CONSTRAINT fk_pedido_fornecedor FOREIGN KEY (id_fornecedor)
        REFERENCES fornecedores(id_fornecedor)
);

-- 9. Pedido_Insumo
CREATE TABLE pedido_insumo (
    id_pedido NUMBER NOT NULL,
    id_insumo NUMBER NOT NULL,
    quantidade NUMBER(6),
    CONSTRAINT pk_pedido_insumo PRIMARY KEY (id_pedido, id_insumo),
    CONSTRAINT fk_pi_pedido FOREIGN KEY (id_pedido)
        REFERENCES pedido(id_pedido),
    CONSTRAINT fk_pi_insumo FOREIGN KEY (id_insumo)
        REFERENCES insumo(id_insumo)
);

-- 10. Movimentacoes
CREATE TABLE movimentacoes ( 
    id_movimentacoes NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    motivo VARCHAR2(20) NOT NULL,
    data_hora_entrada DATE,
    data_hora_saida DATE,
    tipo_movimentacao VARCHAR2(30) NOT NULL,
    quantidade NUMBER(6) NOT NULL,
    id_funcionario NUMBER NOT NULL,
    CONSTRAINT fk_mov_funcionario FOREIGN KEY (id_funcionario)
        REFERENCES funcionarios(id_funcionario),
    CONSTRAINT chk_quantidade CHECK (quantidade > 0),
    CONSTRAINT chk_tipo_movimentacao CHECK (tipo_movimentacao IN ('Entrada', 'Saída')),
    CONSTRAINT chk_motivo CHECK (motivo IN ('Reabastecimento', 'Transferência', 'Vencimento', 'Avaria'))
);

-- 11. Fornecedor_Insumo
CREATE TABLE fornecedor_insumo (
    id_fornecedor NUMBER NOT NULL,
    id_insumo NUMBER NOT NULL,
    CONSTRAINT fk_fornecedor_insumo_fornecedor FOREIGN KEY (id_fornecedor)
        REFERENCES fornecedores(id_fornecedor),
    CONSTRAINT fk_fornecedor_insumo_insumo FOREIGN KEY (id_insumo)
        REFERENCES insumo(id_insumo),
    CONSTRAINT pk_fornecedor_insumo PRIMARY KEY (id_fornecedor, id_insumo)
);

-- 12. Trigger
CREATE OR REPLACE TRIGGER trg_valida_movimentacao
BEFORE INSERT OR UPDATE ON movimentacoes
FOR EACH ROW
DECLARE
    v_cargo funcionarios.cargo%TYPE;
BEGIN
    BEGIN
        SELECT cargo INTO v_cargo
        FROM funcionarios
        WHERE id_funcionario = :NEW.id_funcionario;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20002, 'Funcionário não encontrado.');
    END;

    IF v_cargo = 'Colaborador Interno' AND :NEW.tipo_movimentacao = 'Entrada' THEN
        RAISE_APPLICATION_ERROR(-20001, 'Colaborador Interno só pode realizar Saída.');
    END IF;

    IF :NEW.tipo_movimentacao = 'Entrada' AND :NEW.data_hora_saida IS NOT NULL THEN
        RAISE_APPLICATION_ERROR(-20003, 'Movimentação de Entrada não pode ter data de saída.');
    END IF;

    IF :NEW.tipo_movimentacao = 'Saída' AND :NEW.data_hora_entrada IS NOT NULL THEN
        RAISE_APPLICATION_ERROR(-20004, 'Movimentação de Saída não pode ter data de entrada.');
    END IF;

    IF :NEW.tipo_movimentacao = 'Entrada' AND :NEW.data_hora_entrada IS NULL THEN
        :NEW.data_hora_entrada := SYSDATE;
    ELSIF :NEW.tipo_movimentacao = 'Saída' AND :NEW.data_hora_saida IS NULL THEN
        :NEW.data_hora_saida := SYSDATE;
    END IF;
END;
